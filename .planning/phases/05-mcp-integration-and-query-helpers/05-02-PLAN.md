```elixir
defmodule OrkestraMcp.Tools.GenAggregate do
  @moduledoc "Generate an Orkestra Aggregate module with decide/evolve clauses"

  use Hermes.Server.Component, type: :tool

  schema do
    field(:module_name, :string,
      required: true,
      description: "Full aggregate module name, e.g. MyApp.Orders.OrderAggregate"
    )
    field(:stream_id_field, :string, required: true, description: "...")
    field(:commands, :string, required: true, description: ~s(JSON array...))
    field(:events, :string, required: true, description: ~s(JSON array...))
  end

  @impl true
  def execute(%{module_name: mn, stream_id_field: sf, commands: cj, events: ej}, _frame) do
    project_dir = Application.get_env(:orkestra_mcp, :project_dir)
    commands = Jason.decode!(cj)
    events = Jason.decode!(ej)
    {source, file_path} = OrkestraMcp.Generator.gen_aggregate(mn, sf, commands, events)
    written = OrkestraMcp.Generator.write!(source, project_dir, file_path)
    {:ok, "Created #{written}\n\n```elixir\n#{source}\n```"}
  end
end
```

  - `fields` (required string): ~s(JSON array of fields: [{"name":"order_id","type":"binary_id"},{"name":"status","type":"string"}])
