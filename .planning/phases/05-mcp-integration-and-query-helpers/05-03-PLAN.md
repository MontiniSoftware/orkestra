```elixir
def discover(project_dir) do
  # Scans lib/**/*.ex, returns:
  # %{commands: [...], events: [...], command_handlers: [...],
  #   event_handlers: [...], aggregates: [...]}
end

# Each detect_* follows this pattern:
defp detect_aggregates(acc, content) do
  if content =~ ~r/@behaviour\s+Orkestra\.Aggregate/ do
    case extract_module_name(content) do
      nil -> acc
      module_name -> %{acc | aggregates: acc.aggregates ++ [%{module: module_name}]}
    end
  else
    acc
  end
end

# With Regex.run variant (detect_command_handlers):
defp detect_command_handlers(acc, content) do
  case Regex.run(~r/use\s+Orkestra\.CommandHandler,\s*command:\s*([\w.]+)/, content) do
    [_, command_module] ->
      case extract_module_name(content) do
        nil -> acc
        module_name ->
          entry = %{module: module_name, command: command_module}
          %{acc | command_handlers: acc.command_handlers ++ [entry]}
      end
    nil -> acc
  end
end

defp extract_module_name(content) do
  case Regex.run(~r/defmodule\s+([\w.]+)/, content) do
    [_, name] -> name
    nil -> nil
  end
end

# parse_file pipes through all detect_* functions:
defp parse_file(content, acc) do
  acc
  |> detect_commands(content)
  |> detect_events(content)
  |> detect_command_handlers(content)
  |> detect_event_handlers(content)
  |> detect_aggregates(content)
end

# build_domain_map uses discover/1 and builds text lines:
def build_domain_map(project_dir) do
  %{commands: commands, events: events, command_handlers: ch,
    event_handlers: eh, aggregates: aggregates} = discover(project_dir)
  # ... builds lines with "(command)", "(event)", "(aggregate)" labels ...
end
```

```elixir
defmodule OrkestraMcp.Resources.ListAggregates do
  @moduledoc "Lists all Orkestra Aggregate modules in the project"

  use Hermes.Server.Component,
    type: :resource,
    uri: "orkestra://aggregates",
    mime_type: "application/json"

  @impl true
  def read(_params, _frame) do
    project_dir = Application.get_env(:orkestra_mcp, :project_dir)
    %{aggregates: aggregates} = OrkestraMcp.Introspection.discover(project_dir)
    {:ok, Jason.encode!(aggregates, pretty: true)}
  end
end
```

```elixir
defp detect_projectors(acc, content) do

      case extract_module_name(content) do
```
