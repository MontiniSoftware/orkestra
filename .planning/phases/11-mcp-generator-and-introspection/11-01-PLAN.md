```elixir
defmodule OrkestraMcp.Tools.GenEsQueries do
  use Hermes.Server.Component, type: :tool
  schema do
    field(:module_name, :string, required: true, description: "...")
    field(:projector_module, :string, required: true, description: "...")
  end
  @impl true
  def execute(%{module_name: m, projector_module: p}, _frame) do
    project_dir = Application.get_env(:orkestra_mcp, :project_dir)
    {source, file_path} = OrkestraMcp.Generator.gen_es_queries(m, p)
    written = OrkestraMcp.Generator.write!(source, project_dir, file_path)
    {:ok, "Created #{written}\n\n```elixir\n#{source}\n```"}
  end
end
```
