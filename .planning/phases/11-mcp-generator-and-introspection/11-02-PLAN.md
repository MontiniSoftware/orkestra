defp detect_projectors(acc, content) do
  case Regex.run(~r/use\s+Orkestra\.Projector,\s*repo:\s*([\w.]+)/, content) do
    [_, repo_module] ->
      case extract_module_name(content) do
        nil -> acc

          entry = %{module: module_name, repo: repo_module, events: events}

  end
